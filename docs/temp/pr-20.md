import { test, expect, Page, APIRequestContext } from "@playwright/test";

const BASE_URL = process.env.QA_BASE_URL ?? "https://cnsc.profemarlon.com";

const GUIDED_ACTIONS = [
  "Dame una pista",
  "Explícame esta pregunta",
  "Compara las opciones sin decir cuál es la correcta",
  "Analiza mi justificación",
  "Explícame el feedback",
  "Qué tema debo reforzar",
];

async function gotoPractice(page: Page) {
  await page.goto(`${BASE_URL}/practice`, { waitUntil: "domcontentloaded" });
  await page.waitForLoadState("networkidle");
}

async function gotoDashboard(page: Page) {
  await page.goto(`${BASE_URL}/dashboard`, { waitUntil: "domcontentloaded" });
  await page.waitForLoadState("networkidle");
}

async function answerCurrentQuestion(page: Page) {
  const radioLikeOptions = page.locator(
    'input[type="radio"], button[data-option-key], [role="radio"]'
  );

  const count = await radioLikeOptions.count();
  if (count === 0) {
    throw new Error("No encontré opciones seleccionables en práctica.");
  }

  await radioLikeOptions.first().click({ force: true });

  const rationale = page.locator("textarea").filter({ hasNotText: "Tutor GCM" }).first();
  if (await rationale.count()) {
    await rationale.fill("Seleccioné esta opción para avanzar la validación del tutor.");
  }

  const submitCandidates = [
    page.getByRole("button", { name: /responder|enviar respuesta|confirmar|continuar/i }),
    page.locator('button[type="submit"]'),
  ];

  let submitted = false;
  for (const candidate of submitCandidates) {
    const n = await candidate.count();
    if (n > 0) {
      await candidate.first().click({ force: true });
      submitted = true;
      break;
    }
  }

  if (!submitted) {
    throw new Error("No encontré botón para responder la pregunta.");
  }

  await page.waitForLoadState("networkidle");
}

async function ensureTutorVisible(page: Page) {
  const panel = page.getByTestId("tutor-gcm-panel");
  await expect(panel).toBeVisible();
  await expect(page.getByTestId("tutor-gcm-form")).toBeVisible();
  await expect(page.getByTestId("tutor-gcm-message")).toBeVisible();
  await expect(page.getByTestId("tutor-gcm-submit")).toBeVisible();
}

async function checkGuidedActions(page: Page) {
  for (const action of GUIDED_ACTIONS) {
    await expect(page.getByRole("button", { name: action })).toBeVisible();
  }
}

test.describe("Tutor GCM — últimos sprints", () => {
  test("audita tutor, guardrails, borrador libre, trazas y dashboard", async ({
    page,
    request,
  }) => {
    await gotoPractice(page);
    await ensureTutorVisible(page);

    // Sprint 16/18/20: acciones guiadas visibles y copy consolidado
    await checkGuidedActions(page);

    // Sprint 20: preservación del borrador libre
    const tutorTextarea = page.getByTestId("tutor-gcm-message");
    const draftText = "Tengo una duda libre que no quiero perder.";
    await tutorTextarea.fill(draftText);

    await page.getByRole("button", { name: "Dame una pista" }).click();
    await page.waitForLoadState("networkidle");

    await expect(tutorTextarea).toHaveValue(draftText);

    // Sprint 9-20: guardrail antes de responder
    const tutorResponseBefore = page.locator('[data-testid="tutor-gcm-panel"]').locator("text=/respuesta correcta|clave registrada|opción correcta/i");
    const panelTextBefore = await page.getByTestId("tutor-gcm-panel").innerText();

    expect(
      /no puedo revelar|no revelo la clave|sin revelar la clave|todavía no puedo revelar/i.test(panelTextBefore)
    ).toBeTruthy();

    // Sprint 15 / 15.1: endpoint de trazas responde autenticado
    const preTraceResponse = await request.get(`${BASE_URL}/api/tutor/traces/summary`);
    expect(preTraceResponse.status()).toBe(200);

    const preTraceJson = await preTraceResponse.json();
    expect(preTraceJson).toHaveProperty("totalTurns");
    expect(preTraceJson).toHaveProperty("degradedTurns");
    expect(preTraceJson).toHaveProperty("preAnswerGuardrailHits");
    expect(preTraceJson).toHaveProperty("postAnswerExplanations");
    expect(Array.isArray(preTraceJson.topIntents)).toBeTruthy();
    expect(Array.isArray(preTraceJson.topGuardrails)).toBeTruthy();

    // Responder una pregunta para habilitar explicación post-respuesta
    await answerCurrentQuestion(page);

    // Sprint 11 en adelante: explicación post-respuesta
    await expect(page.getByRole("button", { name: "Explícame el feedback" })).toBeVisible();
    await page.getByRole("button", { name: "Explícame el feedback" }).click();
    await page.waitForLoadState("networkidle");

    const panelTextAfter = await page.getByTestId("tutor-gcm-panel").innerText();
    expect(
      /clave registrada|opción correcta|feedback oficial|distractores/i.test(panelTextAfter)
    ).toBeTruthy();

    // Sprint 15 / 15.1: endpoint sigue sano después de interacción real
    const postTraceResponse = await request.get(`${BASE_URL}/api/tutor/traces/summary`);
    expect(postTraceResponse.status()).toBe(200);

    const postTraceJson = await postTraceResponse.json();
    expect(postTraceJson.totalTurns).toBeGreaterThanOrEqual(0);
    expect(Array.isArray(postTraceJson.topIntents)).toBeTruthy();
    expect(Array.isArray(postTraceJson.topGuardrails)).toBeTruthy();

    // Sprint 17: consumidor visible en dashboard
    await gotoDashboard(page);

    await expect(page.getByText(/Tutor GCM/i)).toBeVisible();
    await expect(page.getByText(/Resumen de uso reciente/i)).toBeVisible();
    await expect(page.getByText(/Solo lectura/i)).toBeVisible();

    const dashboardText = await page.locator("body").innerText();
    expect(/Total de turnos/i.test(dashboardText)).toBeTruthy();
    expect(/Turnos degradados/i.test(dashboardText)).toBeTruthy();
    expect(/Guardrails pre-respuesta/i.test(dashboardText)).toBeTruthy();
    expect(/Explicaciones post-respuesta/i.test(dashboardText)).toBeTruthy();

    // Logout y protección posterior
    const logoutButton = page.getByRole("button", { name: /cerrar sesión|logout|salir/i });
    if (await logoutButton.count()) {
      await logoutButton.first().click({ force: true });
      await page.waitForLoadState("networkidle");

      await page.goto(`${BASE_URL}/practice`, { waitUntil: "domcontentloaded" });
      await page.waitForLoadState("networkidle");

      expect(page.url()).not.toContain("/practice");
    }
  });
});
